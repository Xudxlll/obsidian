## 核心理解

清洗的目标是减少噪声，而不是尽可能删掉字符。课堂三个版本从多格式读取、基础正则，逐渐增加哈希去重、TF-IDF 相似度和配置化流程。

精确去重可以按规范化后的文本哈希完成；近似去重则比较段落向量或词项相似度，用于识别轻微改写、页眉页脚重复等情况。阈值决定删除范围，必须用真实样例检查。

例如两条合同只有“可以”与“不可以”不同，词面非常相似，业务含义却相反。近似重复不能仅凭高相似度一律删除。更稳妥的做法是记录原文、清洗文本、删除原因和来源映射。

清洗与切片也相互影响：过早删除换行可能毁掉标题层级，合并段落会让后续切片跨越不相关内容。先确认文档结构，再选择规则。

## 课堂代码怎么读

按三个版本阅读：`ext_text/clean_text` 负责抽取与基础处理，第二版做段落去重和相似度阈值，第三版加入 Unicode、HTML、配置与多加载器。比较每一步处理前后具体改变了哪些字符。

> [!warning] 易错点与资料边界
> 第三版 `from config import *` 依赖未附在笔记中的配置包；阈值 0.6、0.85 等是课堂实验值，不能脱离数据照搬。全量两两相似度会随段落数增加而快速占用内存。

> [!quote]- 大模型全栈（中）.md · 文本清洗3

> **文本清洗3**
>
> ```python
> """
> 文本的综合清洗
> """
> import os
> import re
> import hashlib
> import unicodedata
> from bs4 import BeautifulSoup
> from sklearn.feature_extraction.text import TfidfVectorizer
> from sklearn.metrics.pairwise import cosine_similarity
> import numpy as np
> from collections import defaultdict
> from langchain_community.document_loaders import TextLoader, PyMuPDFLoader, Docx2txtLoader
> from config import *
>
> os.makedirs(OUTPUT_DIR, exist_ok=True)
>
> # 文档加载
> LOADERS = {
>     ".pdf": lambda f: "\n".join(d.page_content for d in PyMuPDFLoader(f).load()),
>     ".docx": lambda f: "\n".join(d.page_content for d in Docx2txtLoader(f).load()),
>     ".txt": lambda f: TextLoader(f, encoding="utf-8").load()[0].page_content,
>     ".md": lambda f: TextLoader(f, encoding="utf-8").load()[0].page_content
> }
> raw_texts = {}
> for filename in sorted(os.listdir(INPUT_DIR)):
>     filepath = os.path.join(INPUT_DIR, filename)
>     if os.path.isfile(filepath):
>         ext = os.path.splitext(filename)[1].lower()
>         if ext in LOADERS:
>             raw_texts[filename] = LOADERS[ext](filepath)
>
> # 基础清洗
> def clean_text(text):
>     # HTML清洗
>     while re.search(HTML_TAG_PATTERN, text):
>         text = BeautifulSoup(text, "html.parser").get_text(separator="\n")
>
>     # unicode归一化
>     text = unicodedata.normalize("NFKC", text)
>
>     # 统一换行
>     text = text.replace("\r\n", "\n").replace("\r", "\n")
>
>     # 过滤 C类控制字符
>     text = "".join(c for c in text if unicodedata.category(c)[0] != "C" or c in "\n\t")
>
>     # 乱码
>     for pattern, _ in GARBLED_PATTERNS:
>         text = re.sub(pattern, "", text)
>
>     # 压缩空格
>     text = "\n".join(re.sub(r' {2,}', ' ', line).strip() for line in text.split("\n"))
>
>     # 压缩连续换行
>     text = re.sub(r"\n{3,}", "\n\n", text)
>
>     # 页眉+页脚+水印
>     for pattern in HEADER_FOOTER_PATTERNS + WATERMARK_PATTERNS:
>         text = re.sub(pattern, '', text, flags=re.IGNORECASE | re.MULTILINE)
>
>     # Markdown符号
>     for pattern, replacement, user_multiline, _ in MARKDOWN_PATTERNS:
>         text = re.sub(pattern, replacement, text, flags=re.MULTILINE if user_multiline else 0)
>
>     text = text.replace(*MARKDOWN_INLINE_PIPE)
>
>     # 孤立的页码,无效的分割线
>     for pattern in NOISE_LINE_PATTERNS:
>         text = re.sub(pattern, "", text, flags=re.MULTILINE)
>
>     return re.sub(r"\n{3,}", "\n\n", text).strip()
>
> cleaed_texts = {fn: clean_text(t) for fn, t in raw_texts.items()}
>
> # 段落切分
> all_paragraphs = [
>     (fn, para.strip())
>     for fn, t in cleaed_texts.items()
>     for para in t.split("\n")
>     if len(para.strip()) >= MIN_PARAGRAPH_LENGTH
> ]
>
> # MD5去重
> seen = set()
> after_hash = []
> for source, para in all_paragraphs:
>     h = hashlib.md5(re.sub(r'\s+', '', para).encode("utf-8")).hexdigest()
>
>     if h not in seen:
>         seen.add(h)
>         after_hash.append((source, para))
> hash_removed = len(all_paragraphs) - len(after_hash)
>
> # 模糊去重
> final_paragraphs = after_hash
> fuzzy_removed = 0
> if len(after_hash) > 1:
>     sources = [s for s, _ in after_hash]  # [1.pdf ,2.txt...]
>     paras = [p for _, p in after_hash]  # [文本1,文本2...]
>     vec = TfidfVectorizer(analyzer="char", ngram_range=(2, 3), max_features=20000)
>     sim = cosine_similarity(vec.fit_transform(paras))
>
>     keep = np.ones(len(paras), dtype=bool)
>     for i in range(len(paras)):
>         if keep[i]:
>             keep[i + 1:] &= sim[i, i + 1:] < SIMILARITY_THRESHOLD
>
>     final_paragraphs = [(sources[i], paras[i]) for i in range(len(paras)) if keep[i]]
>
>     fuzzy_removed = len(after_hash) - len(final_paragraphs)
>
> # 质量的校验
> total_chars = sum(len(p) for _, p in final_paragraphs)
> short_count = sum(1 for _, p in final_paragraphs if len(p) < MIN_PARAGRAPH_LENGTH)
> long_count = sum(1 for _, p in final_paragraphs if len(p) > MAX_PARAGRAPH_LENGTH)
> normal_ratio = (len(final_paragraphs) - short_count - long_count) / len(final_paragraphs) * 100
> noise_count = sum(
>     1 for _, p in final_paragraphs
>     if any(re.search(pat, p) for pat in NOISE_LINE_PATTERNS)
> )
> noise_rate = noise_count / len(final_paragraphs) * 100
>
> print(
>     f"总字数量:{total_chars},短段落:{short_count},长段落{long_count},正常段落占比:{normal_ratio},噪声段落占比:{noise_rate}")
>
> # 保存
> groped = defaultdict(list)
> for source, para in final_paragraphs:
>     groped[source].append(para)
>
> for source, paras in groped.items():
>     basename = os.path.splitext(source)[0]
>     with open(os.path.join(OUTPUT_DIR, f"{basename}_cleaned.txt"), "w", encoding="utf-8") as f:
>         f.write("\n\n".join(paras) + "\n\n")
> ```
