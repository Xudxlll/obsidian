> [!abstract] 学习目标
> 把模型能提出什么动作，与程序允许什么动作分开。

## 核心理解

课堂示例用 Tool 元数据、角色权限、参数校验、危险开关与人工确认控制工具调用，再把允许与拒绝都写进审计日志。这说明约束应落在执行层，不能只写在提示词里。

路径校验先解析真实路径，再确认位于指定根目录，防止相对路径越界。计算器用字符白名单限制输入，但仍需限制表达式长度与资源，避免合法字符组成极端耗时任务。

审计的哈希链能帮助发现部分内容变更，前提是链头或外部锚点可信；若攻击者能重写整份日志和所有哈希，就不能只靠文件自身证明不可篡改。

代码执行示例使用子进程、超时与部分平台限制。临时目录与超时不等于操作系统级隔离，不能把这段教学程序当作足以执行任意不可信代码的防护产品。

## 课堂代码怎么读

先看 run_tool 的每一道拒绝条件，再看审计记录的前序哈希与当前记录；最后读 sandbox_exec 的启动、超时和平台分支。注意邮件与转账函数明确是模拟返回。

> [!warning] 易错点与资料边界
> Windows Job Object 限制不应推断为在所有系统都有效。代码保留为课程材料，本轮没有执行其中的删除、计算或子进程示例。

## 复盘练习

为什么工具函数内部也要检查参数？

> [!question]- 参考思路
> 模型调用不是唯一入口，测试、其他程序或恢复流程也可能直接调用。关键约束放在实际动作边界，才更容易保持一致。

> ```python
> """
> 安全与风控: 权限控制
> """
>
> import sys
> import os
> import re
> import json
> import time
> import tempfile
> import threading
> import hashlib
> import subprocess
> from pathlib import Path
> from dataclasses import dataclass
> from typing import Callable, Optional
>
> # 1.全局配置
> BASE_DIR = Path(__file__).resolve().parent
> SANDBOX_ROOT = BASE_DIR / "sandbox_workspace"  # 文件工具唯一可访问目录
> SANDBOX_ROOT.mkdir(parents=True, exist_ok=True)
>
> # 角色 -> 可用工具集（最小权限：数据 Agent 没有发邮件工具）
> ROLE_TOOLS: dict[str, set[str]] = {
>     "data_agent": {"calculator", "file_read"},
>     "operator_agent": {"calculator", "file_read", "file_write", "send_email"},
> }
>
> # 危险工具：默认禁用，必须"单独授权"后才可用
> DANGEROUS_TOOLS = {"delete", "transfer", "send_email"}
> # 已单独授权的工具
> AUTHORIZED_TOOLS: set[str] = set()
> # 需要人工审批的工具
> APPROVAL_REQUIRED = {"send_email", "file_write", "delete", "transfer"}
>
> # 审批决策器：生产环境应接入工单/IM/网页审批流,默认拒绝）
> APPROVER: Callable[[str, dict], bool] = lambda name, params: False
>
> # 审计级：哈希链日志（追加写入，每条含前一条哈希，篡改即可被检测）
> class AuditLog:
>     def __init__(self, path: Path):
>         self.path = path
>         self.path.parent.mkdir(parents=True, exist_ok=True)
>         self._prev_hash = self._load_prev()
>         self._lock = threading.Lock()
>
>     def _load_prev(self) -> str:
>         if not self.path.exists():
>             return "0" * 64
>         lines = self.path.read_text(encoding="utf-8").strip().splitlines()
>         if not lines:
>             return "0" * 64
>         try:
>             return json.loads(lines[-1])["hash"]
>         except Exception:
>             return "0" * 64
>
>     def record(self, entry: dict) -> None:
>         with self._lock:
>             rec = {**entry, "ts": round(time.time(), 3), "prev": self._prev_hash}
>             body = json.dumps(rec, ensure_ascii=False, sort_keys=True)
>             rec["hash"] = hashlib.sha256((self._prev_hash + body).encode()).hexdigest()
>             with self.path.open("a", encoding="utf-8") as f:
>                 f.write(json.dumps(rec, ensure_ascii=False) + "\n")
>             self._prev_hash = rec["hash"]
>
> AUDIT = AuditLog(BASE_DIR / "audit.log")
>
> # 参数级：校验器（白名单 / 操作范围）
> _CALC_WHITELIST = re.compile(r"^[0-9+\-*/().\s]+$")
>
> def validate_calc_expr(expr: str) -> None:
>     """白名单校验：表达式只能含数字、运算符、括号、空格。"""
>     if not isinstance(expr, str) or not _CALC_WHITELIST.match(expr):
>         raise ValueError(f"非法表达式（白名单拦截）: {expr!r}")
>
> def validate_in_sandbox(path: str) -> Path:
>     """操作范围限制：解析后的真实路径必须位于 SANDBOX_ROOT 之内。"""
>     real = (SANDBOX_ROOT / path).resolve()
>     if SANDBOX_ROOT.resolve() not in real.parents and real != SANDBOX_ROOT.resolve():
>         raise ValueError(f"越权路径（只允许沙箱目录）: {path!r}")
>     return real
>
> # 工具实现（业务逻辑本身尽量精简）
> def tool_calculator(expr: str) -> str:
>     return str(eval(expr, {"__builtins__": {}}, {}))
>
> def tool_file_read(path: str) -> str:
>     p = validate_in_sandbox(path)
>     return p.read_text(encoding="utf-8")
>
> def tool_file_write(path: str, content: str) -> str:
>     p = validate_in_sandbox(path)
>     p.parent.mkdir(parents=True, exist_ok=True)
>     p.write_text(content, encoding="utf-8")
>     return f"已写入 {p.name}"
>
> def tool_send_email(to: str, body: str) -> str:
>     return f"[模拟发送] -> {to}: {body[:20]}..."
>
> def tool_delete(path: str) -> str:
>     p = validate_in_sandbox(path)
>     p.unlink(missing_ok=True)
>     return f"已删除 {p.name}"
>
> def tool_transfer(amount: float, to: str) -> str:
>     return f"[模拟转账] {amount} -> {to}"
>
> # 工具级 + 执行级 + 审批级：统一调度守卫
> @dataclass
> class Tool:
>     name: str
>     fn: Callable
>     roles: set[str]
>     dangerous: bool = False
>     needs_approval: bool = False
>     validate: Optional[Callable[[dict], None]] = None
>
> REGISTRY: dict[str, Tool] = {
>     "calculator": Tool("calculator", tool_calculator, {"data_agent", "operator_agent"},
>                        validate=lambda p: validate_calc_expr(p["expr"])),
>     "file_read": Tool("file_read", tool_file_read, {"data_agent", "operator_agent"}),
>     "file_write": Tool("file_write", tool_file_write, {"operator_agent"}, needs_approval=True,
>                        validate=lambda p: validate_in_sandbox(p["path"])),
>     "send_email": Tool("send_email", tool_send_email, {"operator_agent"}, dangerous=True, needs_approval=True),
>     "delete": Tool("delete", tool_delete, {"operator_agent"}, dangerous=True, needs_approval=True,
>                    validate=lambda p: validate_in_sandbox(p["path"])),
>     "transfer": Tool("transfer", tool_transfer, {"operator_agent"}, dangerous=True, needs_approval=True),
> }
>
> def run_tool(tool_name: str, role: str, params: dict) -> str:
>     """执行前的五道闸门：角色 -> 危险禁用 -> 参数校验 -> 人工审批 -> 执行 + 审计。"""
>     tool = REGISTRY.get(tool_name)
>     if tool is None:
>         return _deny(tool_name, role, params, "工具不存在")
>     if role not in tool.roles:  # 工具级：角色越权
>         return _deny(tool_name, role, params, f"角色 {role} 无权使用该工具")
>     if tool.dangerous and tool_name not in AUTHORIZED_TOOLS:  # 工具级：危险工具未单独授权
>         return _deny(tool_name, role, params, "危险工具未单独授权（默认禁用）")
>     try:
>         if tool.validate:
>             tool.validate(params)  # 参数级：白名单 / 范围
>     except Exception as e:
>         return _deny(tool_name, role, params, f"参数校验失败: {e}")
>     if tool.needs_approval and not APPROVER(tool_name, params):  # 审批级：强制 HITL
>         return _deny(tool_name, role, params, "人工审批未通过")
>
>     try:
>         result = tool.fn(**params)
>         AUDIT.record({"who": role, "tool": tool_name, "params": params,
>                       "decision": "ALLOW", "result": str(result)[:200]})
>         return result
>     except Exception as e:
>         AUDIT.record({"who": role, "tool": tool_name, "params": params,
>                       "decision": "ERROR", "result": str(e)[:200]})
>         return f"执行异常: {e}"
>
> def _deny(tool_name: str, role: str, params: dict, reason: str) -> str:
>     AUDIT.record({"who": role, "tool": tool_name, "params": params,
>                   "decision": "DENY", "result": reason})
>     return f"[拒绝] {reason}"
>
> # 执行级：沙箱执行
> def _apply_job_limits(proc: subprocess.Popen, mem_mb: int = 256, max_procs: int = 1) -> None:
>     """Windows 专用：用 Job Object 限制子进程内存与进程数。失败则静默降级为仅超时限制。"""
>     if sys.platform != "win32":
>         return
>     try:
>         import ctypes
>         from ctypes import wintypes
>         k32 = ctypes.windll.kernel32
>
>         class IO_COUNTERS(ctypes.Structure):
>             _fields_ = [(n, ctypes.c_ulonglong) for n in (
>                 "ReadOperationCount", "WriteOperationCount", "OtherOperationCount",
>                 "ReadTransferCount", "WriteTransferCount", "OtherTransferCount")]
>
>         class BASIC_LIMIT(ctypes.Structure):
>             _fields_ = [
>                 ("LimitFlags", wintypes.DWORD),
>                 ("MinimumWorkingSetSize", ctypes.c_size_t),
>                 ("MaximumWorkingSetSize", ctypes.c_size_t),
>                 ("ActiveProcessLimit", wintypes.DWORD),
>                 ("PriorityClass", wintypes.DWORD),
>                 ("SchedulingClass", wintypes.DWORD)
>             ]
>
>         class EXT_LIMIT(ctypes.Structure):
>             _fields_ = [
>                 ("BasicLimitInformation", BASIC_LIMIT),
>                 ("IoInfo", IO_COUNTERS),
>                 ("ProcessMemoryLimit", ctypes.c_size_t),
>                 ("JobMemoryLimit", ctypes.c_size_t),
>                 ("PeakProcessMemoryUsed", ctypes.c_size_t),
>                 ("PeakJobMemoryUsed", ctypes.c_size_t)
>             ]
>
>         JOB_LIMIT_JOB_MEMORY = 0x200
>         JOB_LIMIT_ACTIVE_PROC = 0x80
>
>         job = k32.CreateJobObjectW(None, None)
>         info = EXT_LIMIT()
>         info.BasicLimitInformation.LimitFlags = JOB_LIMIT_JOB_MEMORY | JOB_LIMIT_ACTIVE_PROC
>         info.BasicLimitInformation.ActiveProcessLimit = max_procs
>         info.JobMemoryLimit = mem_mb * 1024 * 1024
>         info.ProcessMemoryLimit = mem_mb * 1024 * 1024
>         k32.SetInformationJobObject(job, 9, ctypes.byref(info), ctypes.sizeof(info))
>         k32.AssignProcessToJobObject(job, int(proc._handle))
>     except Exception as e:
>         print(f"[警告] Job Object 资源限制未生效，仅使用超时限制: {e}")
>
> def sandbox_exec(code: str, timeout: int = 10, mem_mb: int = 256) -> str:
>     """在隔离子进程中执行 Python 代码"""
>     with tempfile.NamedTemporaryFile(mode="w", suffix=".py", delete=False,
>                                      dir=str(SANDBOX_ROOT)) as f:
>         f.write(code)
>         tmp_path = f.name
>     try:
>         si = subprocess.STARTUPINFO()
>         si.dwFlags |= subprocess.STARTF_USESHOWWINDOW
>         si.wShowWindow = 0  # SW_HIDE：隐藏子进程窗口
>         restricted_env = {  # 限制环境变量
>             "SYSTEMROOT": os.environ.get("SYSTEMROOT", r"C:\Windows"),
>             "TEMP": str(SANDBOX_ROOT),
>             "TMP": str(SANDBOX_ROOT),
>         }
>         proc = subprocess.Popen(
>             [sys.executable, tmp_path],
>             stdout=subprocess.PIPE, stderr=subprocess.PIPE,
>             cwd=str(SANDBOX_ROOT), env=restricted_env,  # 限制工作目录
>             creationflags=subprocess.CREATE_NO_WINDOW, startupinfo=si, text=True,
>         )
>         _apply_job_limits(proc, mem_mb)  # 执行级：内存/进程数硬限制
>         try:
>             out, err = proc.communicate(timeout=timeout)  # 超时即杀（CPU/时间限制）
>         except subprocess.TimeoutExpired:
>             proc.kill()
>             proc.communicate()
>             return "执行超时（已被终止）"
>         return out if proc.returncode == 0 else f"错误: {err}"
>     finally:
>         os.unlink(tmp_path)
>
