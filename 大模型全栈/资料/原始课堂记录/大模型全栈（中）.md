"""
向量相似度+图 混合式图RAG
向量索引创建->语义召回->图关系扩展->上下文拼接->答案生成
"""
import os
from neo4j import GraphDatabase
from langchain_openai import ChatOpenAI
from langchain_neo4j import Neo4jVector
from langchain_huggingface import HuggingFaceEmbeddings
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from dotenv import load_dotenv

load_dotenv()

class HybridGraphRAG:
    def __init__(self):
        self.driver = GraphDatabase.driver("bolt://localhost:7687", auth=("neo4j", "12345678"))
        self.llm = ChatOpenAI(
            api_key=os.getenv("DEEPSEEK_API_KEY"),
            base_url=os.getenv("DEEPSEEK_BASE_URL"),
            model="deepseek-chat",
            temperature=0
        )
        self.embeddings = HuggingFaceEmbeddings(
            model_name="./bge-base-zh-v1.5",
            model_kwargs={"device": "cpu"},
            encode_kwargs={"normalize_embeddings": True}
        )
        self.vector_index = None

    def build_vector_index(self):
        # 先给业务节点打标签 自动生成词嵌入
        with self.driver.session() as session:
            for label in ["Disease", "Drug", "Symptom"]:
                session.run(f"MATCH (n:{label}) SET n:__Entity__")

        # 自动读取节点,生成向量并建立内部Neo4j索引
        self.vector_index = Neo4jVector.from_existing_graph(
            embedding=self.embeddings,
            url="bolt://localhost:7687",
            username="neo4j",
            password="12345678",
            index_name="entity_embeddings",  # 内部的名字
            node_label="__Entity__",
            text_node_properties=["name"],
            embedding_node_property="embedding",  # 计算出向量存在节点属性中
            search_type="vector"
        )
        print("向量索引构建完成~")

    def semantic_recall(self, question, top_k):
        with self.driver.session() as session:
            query_embedding = self.embeddings.embed_query(question)
            records = session.run("""
                CALL db.index.vector.queryNodes('entity_embeddings', $top_k, $embedding)
                YIELD node, score
                RETURN node.name AS name, labels(node) AS labels, score
                ORDER BY score DESC
            """, top_k=top_k, embedding=query_embedding)

            return [dict(r) for r in records]  # [{"name:xx","labels:xx,"score":...},{},...]

    def graph_expand(self, entity_names):
        if not entity_names:
            return {"nodes": [], "relationships": [], "triples_text": []}
        # 召回实体为起点,遍历多跳邻居,收集子图中的节点和关系作为结构化上下文
        # {"node":[....],"rel":[...],...}
        with self.driver.session() as session:
            # 查询1
            nodes = session.run("""
                UNWIND $seeds AS seeds
                MATCH (seed_node {name:seed})
                MATCH path=(seed_node)-[*1..2]-(neighbor)
                UNWIND nodes(path) AS n
                RETURN DISTINCT n.name AS name, labels(n) AS labels
                LIMIT 20
            """, seeds=entity_names)

            # 查询2
            # 取出路径中的全部关系,得到头尾实体+关系类型+关系属性
            relationships = session.run("""
                 UNWIND $seeds AS seeds
                 MATCH (seed_node {name:seed})
                 MATCH path = (seed_node)-[*1..2]-(neighbor)
                 UNWIND relationships(path) AS r
                 RETURN DISTINCT startNode(r).name AS head,
                        type(r) AS rel,
                        endNode(r).name AS tail,
                        properties(r) AS props
                 LIMIT 20
            """, seeds=entity_names)

        triples_lines = []
        for r in relationships:
            props_str = ""
            if r["props"]:
                props_str = f"{{{','.join(f'{k}:{v}' for k, v in r['props'].items())}}}"

            triples_lines.append(f" {r['head']} --{r['rel']}{props_str}--> {r['tail']}")

        triples_text = "\n".join(triples_lines)
        return {"nodes": nodes, "relationships": relationships, "triples_text": triples_text}

                    CALL apoc.create.node([label], {name: name}) YIELD node
                    RETURN node
                """, entity_list=entity_list)

from langchain_neo4j import Neo4jGraph
from langchain_openai import ChatOpenAI
import os

from langchain_core.output_parsers import StrOutputParser
from dotenv import load_dotenv

load_dotenv()

    database="neo4j"
)

llm = ChatOpenAI(
    api_key=os.getenv("DEEPSEEK_API_KEY"),
    base_url=os.getenv("DEEPSEEK_BASE_URL"),
    model="deepseek-chat",
    temperature=0
)

# 自定义 Cypher Prompt
CYPHER_GENERATION = """
你是一个 Neo4j Cypher 查询专家。根据用户的自然语言问题和图谱 Schema，生成正确的 Cypher 查询语句。
【图谱 Schema】
{schema}
【规则】
1. 只生成只读查询（MATCH/WITH/RETURN），禁止生成 CREATE/SET/DELETE/MERGE 等写操作
2. 使用正确的标签、关系类型和属性名，必须与 Schema 完全一致
3. 字符串匹配使用 CONTAINS 或 =，不要使用 LIKE
4. 分页使用 SKIP 和 LIMIT
5. 不要在 Cypher 中使用注释
6. 如果问题无法用当前 Schema 回答，返回 "无法回答"
7. 只返回 Cypher 语句本身，不要包含任何解释或 Markdown 代码块标记
【用户问题】
{question}
【生成的 Cypher】
"""
cypher_prompt = PromptTemplate(
    input_variable=["schema", "question"],
    template=CYPHER_GENERATION,
)

cypher_chain = cypher_prompt | llm | StrOutputParser()

# 自定义问答
def ask(question):
    print(f"问题:{question}")
    # 生成 cypher
    cypher = cypher_chain.invoke({
        "schema": graph.schema,
        "question": question
    })
    cypher = re.sub(r"```cypher\s*", "", cypher)
    cypher = re.sub(r"```\s*", "", cypher)
    print(f"生成 cypher:{cypher}")
    if "无法回答" in cypher:
        return {
            "answer": "抱歉,当前知识图谱无法回答这个问题",
            "cypher": None, "result": None
        }

    # 出错自动重试
    max_retries = 2
    res_cypher = None
    for attempt in range(max_retries + 1):
        try:
            # 安全执行
            dangerous_keywords = ["CREATE", "SET", "DELETE", "MERGE", "DROP", "ALTER", "CALL", "REMOVE"]
            cypher_upper = cypher.upper()
            for kw in dangerous_keywords:
                if kw == "CALL" and "DB." in cypher_upper:
                    continue
                if kw in cypher_upper:
                    raise ValueError(f"安全拦截:禁止执行包含 {kw} 的查询")
            res_cypher = graph.query(cypher)
            print(f"查询结果:{res_cypher}")
            break
        except Exception as e:
            print(f"执行失败(尝试{attempt + 1}/{max_retries + 1}):{e}")
            if attempt < max_retries:
                # 把报错信息附加到问题,让LLM修正cypher
                cypher = cypher_chain.invoke({
                    "schema": graph.schema,
                    "question": f"{question}\n\n之前生成的 Cypher 执行报错:{e}\n请修正后重新生成"
                })
                cypher = re.sub(r"```cypher\s*", "", cypher).replace("```", "").strip()
            else:
                return {"answer": f"查询失败:{e}", "cypher": None, "result": None}

    # 生成答案
    answer_prompt = PromptTemplate(
        input_variable=["question", "cypher", "result"],
        template="""你是一个知识图谱问答助手。根据用户问题、执行的 Cypher 和查询结果，生成简洁准确的自然语言回答。
【用户问题】{question}
【执行的 Cypher】{cypher}
【查询结果】{result}
【要求】
1. 直接回答问题，不要提及 Cypher 或查询过程
2. 如果结果为空，说明未找到相关信息
3. 数据要准确，不要编造结果中没有的信息
4. 回答简洁明了
【回答】
        """)
    answer_chain = answer_prompt | llm | StrOutputParser()
    answer = answer_chain.invoke({
        "question": question,
        "cypher": cypher,
        "result": str(res_cypher)
    })

    return {"answer": answer, "cypher": cypher, "result": res_cypher}

# 测试
questions = [

import os

    url="bolt://localhost:7687",
    username="neo4j",
    password="12345678",
    database="neo4j"
)

    api_key=os.getenv("DEEPSEEK_API_KEY"),
    base_url=os.getenv("DEEPSEEK_BASE_URL"),
    model="deepseek-chat",
    temperature=0
)

# 构建链路: 自然语言输入->LLM生成Cypher语句->执行查询->整理结果
chain = GraphCypherQAChain.from_llm(
    llm=llm,
    graph=graph,
    verbose=True,
    allow_dangerous_requests=True,
    return_intermediate_steps=True
)

question = "北京和深圳有多少人?"
response = chain.invoke({"query": question})

print(response["result"],"\n")

# 中间步骤
for step in response["intermediate_steps"]:
    if "query" in step:
        print(f"哈?step:{step['query']}")
    if "context" in step:
        print(f"结果:{step['context']}")

# 属性相似度

```
"""
实体对齐  属性相似度 + 大模型
"""
import re
import numpy as np
from rapidfuzz import fuzz

# 数据
entities = [
    {"name": "阿里巴巴集团", "attrs": {"成立年份": "1999", "总部": "杭州", "创始人": "马云"}, "source": "A"},
    {"name": "Alibaba Group", "attrs": {"foundedYear": "1999", "headquarters": "Hangzhou", "founder": "Jack Ma"},
     "source": "B"},
    {"name": "阿里", "attrs": {"成立年份": "1999", "总部": "杭州"}, "source": "C"},
    {"name": "腾讯", "attrs": {"成立年份": "1998", "总部": "深圳", "创始人": "马化腾"}, "source": "A"},
    {"name": "Tencent", "attrs": {"foundedYear": "1998", "headquarters": "Shenzhen"}, "source": "B"},
    {"name": "苹果公司", "attrs": {"成立年份": "1976", "总部": "库比蒂诺"}, "source": "A"},
    {"name": "Apple Inc.", "attrs": {"foundedYear": "1976", "headquarters": "Cupertino"}, "source": "B"},
    {"name": "苹果", "attrs": {"类型": "水果", "产地": "山东"}, "source": "C"},
    {"name": "苹果手机", "attrs": {"手机类型": "Iphone17", "产地": "库比蒂诺"}, "source": "A"},
]

# 属性相似度
clusters = []
for ent in entities:
    if not clusters:
        clusters.append([ent])
        continue

    sim_scores = []

    for cluster in clusters:
        e2 = cluster[0]

        # 归一化
        name1_norm = re.sub(r'[\s\-_·.，,。！!？?()（）]', '', str(ent["name"]).strip().lower())
        name2_norm = re.sub(r'[\s\-_·.，,。！!？?()（）]', '', str(e2["name"]).strip().lower())
        name_sim = fuzz.WRatio(name1_norm, name2_norm) / 100  # 归一化[0,1]

        # 属性键模糊匹配 + 属性值相似度
        attr_sim = 0
        if ent.get("attrs") and e2.get("attrs"):
            matched_keys = []

            for k1 in ent["attrs"]:
                k1_norm = re.sub(r'[\s\-_·.，,。！!？?()（）]', '', str(k1).strip().lower())
                for k2 in e2["attrs"]:
                    k2_norm = re.sub(r'[\s\-_·.，,。！!？?()（）]', '', str(k2).strip().lower())

                    if fuzz.ratio(k1_norm, k2_norm) / 100 > 0.8:
                        matched_keys.append((k1, k2))

            # 属性值
            if matched_keys:
                value_sims = []
                for k1, k2 in matched_keys:
                    v1 = str(ent["attrs"][k1])
                    v2 = str(e2["attrs"][k2])
                    value_sims.append(fuzz.ratio(v1, v2) / 100)

                attr_sim = np.mean(value_sims)

            # 融合
            total_sim = 0.6 * name_sim + 0.4 * attr_sim
            sim_scores.append(total_sim)

    best_idx = int(np.argmax(sim_scores))
    best_sim = sim_scores[best_idx]
    if best_sim >= 0.6:
        clusters[best_idx].append(ent)
    else:
        clusters.append([ent])

# 打印结果
clusters_result = {f"cluster_{i}": cluster for i, cluster in enumerate(clusters)}
for cid, cluster in clusters_result.items():
    print(f"{cid}:{[e['name'] for e in cluster]}")

"""
使用BERT完成NER
原始标注数据集 --> 分词 + 标签对齐 --> 加载模型 --> 设置参数 --> 循环训练  --> 验证集 + 测试评估 --> 保存模型和分词器

"""
import numpy as np
from datasets import load_from_disk
from transformers import AutoTokenizer, AutoModelForTokenClassification, TrainingArguments, Trainer, \
    DataCollatorForTokenClassification
import evaluate

# 数据集
dataset = load_from_disk("./data/ner_dataset")
tokenizer = AutoTokenizer.from_pretrained("./bert-base-chinese")

'''
样本
data_train = [
    {"token":["张","三","去","北","京"],"ner_tags":[1,2,0,8,10]},
    {"token":["马","云","阿","里","巴","巴"],"ner_tags":[1,2,0,3,4,4]},
]
'''

# 分词 子词标签对齐
def tokenizer_labels(examples):
    # tokens = ["张","三"] ner_tags = [1,2]
    # 张 --> B-PER(1) 三 --> i-PER(2)
    tokenized_inputs = tokenizer(
        examples["tokens"],
        truncation=True,
        max_length=128,
        is_split_into_words=True,
    )

    all_labels = []

    for batch_idx, label in enumerate(examples["ner_tags"]):
        # 每个原始词对应下标
        word_ids = tokenized_inputs.word_ids(batch_index=batch_idx)
        prev_word_idx = None
        label_ids = []
        for word_idx in word_ids:
            if word_idx is None:
                label_ids.append(-100)
            elif word_idx != prev_word_idx:
                label_ids.append(label[word_idx])  # [None,0,1,None]
            else:
                label_ids.append(-100)

            prev_word_idx = word_idx
        all_labels.append(label_ids)
    tokenized_inputs["labels"] = all_labels

    return tokenized_inputs

# 转换
tokenized_ds = dataset.map(tokenizer_labels, batched=True)

# 模型
model = AutoModelForTokenClassification.from_pretrained(
    "./bert-base-chinese",
    num_labels=7,
    id2label={0: "O", 1: "B-PER", 2: "I-PER", 3: "B-ORG", 4: "I-ORG", 5: "B-LOC", 6: "I-LOC"},
    label2id={"O": 0, "B-PER": 1, "I-PER": 2, "B-ORG": 3, "I-ORG": 4, "B-LOC": 5, "I-LOC": 6}
)
seqeval_metric = evaluate.load("seqeval")  # 加载 seqeval 指标 (没有库就下载)

# 训练
training_args = TrainingArguments(
    output_dir="./ner_output",  # 输出目录，存放checkpoint
    learning_rate=2e-5,  # 学习率，bert微调常用2e‑5
    per_device_train_batch_size=16,  # 训练每卡batch大小16
    per_device_eval_batch_size=16,  # 验证每卡batch大小16
    num_train_epochs=3,  # 训练轮数3轮
    weight_decay=0.01,  # 权重衰减，防止过拟合
    eval_strategy="epoch",  # 每训练完1轮做一次验证评估
    save_strategy="epoch",  # 每训练完1轮保存checkpoint
    logging_strategy="epoch",  # 每轮输出日志
    load_best_model_at_end=True,  # 训练结束自动加载验证集指标最优的模型
    metric_for_best_model="f1",  # 使用f1判断哪个模型最优
    report_to="none",  # 关闭wandb等第三方日志上报
    fp16=False  # 关闭半精度混合训练
)
data_collator = DataCollatorForTokenClassification(tokenizer=tokenizer)

# 评估函数 eval_pred=(模型的输出logits,真实标签)
def compute_metrics(eval_pred):
    predictions, labels = eval_pred
    predictions = np.argmax(predictions, axis=2)

    # 预测
    true_preds = [
        [{0: "O", 1: "B-PER", 2: "I-PER", 3: "B-ORG", 4: "I-ORG", 5: "B-LOC", 6: "I-LOC"}[p] for p, l in zip(pred, label) if l != -100]
        for pred, label in zip(predictions, labels)
    ]

    # 真实
    true_labels = [
        [{0: "O", 1: "B-PER", 2: "I-PER", 3: "B-ORG", 4: "I-ORG", 5: "B-LOC", 6: "I-LOC"}[l] for p, l in zip(pred, label) if l != -100]
        for pred, label in zip(predictions, labels)
    ]

    result = seqeval_metric.compute(predictions=true_preds, references=true_labels)
    return {
        "precision": result["overall_precision"],
        "recall": result["overall_recall"],
        "f1": result["overall_f1"],
        "accuracy": result["overall_accuracy"],
    }

trainer = Trainer(
    model=model,
    args=training_args,
    train_dataset=tokenized_ds["train"],
    eval_dataset=tokenized_ds["validation"],
    data_collator=data_collator,
    compute_metrics=compute_metrics,
)

if __name__ == "__main__":
    trainer.train()
    eval_result = trainer.evaluate()

    for k,v in eval_result.items():
        print(f"{k}: {v}")

    test_result = trainer.evaluate(tokenized_ds['test'])
    for k,v in test_result.items():
        print(f"{k}: {v}")

    model.save_pretrained("./ner_output/best_model")
    tokenizer.save_pretrained("./ner_output/best_model")

# 6.1 多查询改写 + BM25检索 + 向量检索 + RRF融合
def hybrid_retrieve(question):
    raw = multi_query_chain.invoke({"question": question, "count": 3})
    try:
        queries = json.loads(raw)
        if len(queries) == 0 or not isinstance(queries, list):
            queries = [question]
    except Exception as e:
        queries = [question]

    if question not in queries:
        queries.insert(0, question)

    # 对每个查询做混合检索,合并信息
    all_candidates = {}
    for q in queries:
        # 向量检索
        vec_results = vectorstore.similarity_search_with_score(q, k=25)
        vec_ranking = []
        # 遍历检索的结果,得到doc对象,找到该doc在全局chunks切片列表里的下标索引
        for doc, _ in vec_results:
            for i, c in enumerate(chunks):
                if c.page_content == doc.page_content:
                    vec_ranking.append(i)
                    break
        # BM检索
        bm25_scores = bm25.get_scores(list(jieba.cut(q)))
        bm25_rank = list(np.argsort(bm25_scores)[::-1][:25])

        # RRF融合
        fused = {}
        for ranking in [vec_ranking, bm25_rank]:
            for rank, idx in enumerate(ranking):
                fused[idx] = fused.get(idx, 0) + 1 / (60 + rank + 1)

        # 多查询合并候选集,去重
        for idx, score in fused.items():
            # key 文本 value元组 (文档对象,累计总分数)
            key = chunks[idx].page_content
            if key not in all_candidates:
                all_candidates[key] = (chunks[idx], 0)

# 2. 定义评估函数
def evaluate_config(chunk_size, chunk_overlap, top_k):
    """评估一组超参数配置的检索质量"""
    splitter = RecursiveCharacterTextSplitter(
        chunk_size=chunk_size, chunk_overlap=chunk_overlap,
        separators=["\n\n", "\n", "。", "！", "？", " ", ""],
    )
    chunks = splitter.split_documents(knowledge_docs)
    if len(chunks) == 0:
        return {"recall@5": 0, "ndcg@5": 0, "mrr": 0}

    vectorstore = Chroma.from_documents(chunks, embeddings)

    tokenized = [list(jieba.cut(c.page_content)) for c in chunks]
    bm25 = BM25Okapi(tokenized, k1=1.5, b=0.75)

    qrels_dict, run_dict = {}, {}
    for i, item in enumerate(val_data):
        qid = f"q{i}"
        question = item["question"]
        evidence = item.get("evidence", "")

        vec_results = vectorstore.similarity_search_with_score(question, k=top_k)
        bm25_scores = bm25.get_scores(list(jieba.cut(question)))
        bm25_top = np.argsort(bm25_scores)[::-1][:top_k]

        vec_ranking = []
        for doc,_ in vec_results:
            for idx,c in enumerate(chunks):
                if c.page_content == doc.page_content:
                    vec_ranking.append(idx)
                    break

        bm25_ranking = list(bm25_top)
        fused = {}
        for ranking in [vec_ranking, bm25_ranking]:
            for rank, idx in enumerate(ranking):
                fused[idx] = fused.get(idx, 0) + 1 / (60 + rank + 1)

        run_dict[qid] = {}
        qrels_dict[qid] = {}

        # 用 jieba 分词提取 evidence 关键词，过滤单字停用词
        ev_tokens = set(jieba.cut(evidence)) if evidence else set()
        ev_tokens = {t for t in ev_tokens if len(t) > 1 and t.strip()}

        for rank, (idx, score) in enumerate(sorted(fused.items(), key=lambda x: x[1], reverse=True)[:top_k]):
            if idx < len(chunks):
                doc_id = f"d_{idx}"
                run_dict[qid][doc_id] = float(score)
                content = chunks[idx].page_content

                # 用重叠比例而非精确匹配判断相关性
                if ev_tokens:
                    content_tokens = set(jieba.cut(content))
                    content_tokens = {t for t in content_tokens if len(t) > 1 and t.strip()}
                    overlap = len(ev_tokens & content_tokens)
                    overlap_ratio = overlap / len(ev_tokens)
                    rel = (3 if overlap_ratio >= 0.7 else
                           2 if overlap_ratio >= 0.4 else
                           1 if overlap_ratio >= 0.15 else 0)
                else:
                    rel = 0

                if rel > 0:
                    qrels_dict[qid][doc_id] = rel

        # 兜底，确保每个 query 至少有一个相关文档，避免 ranx 报错==
        if not qrels_dict[qid] and run_dict[qid]:
            first_doc = list(run_dict[qid].keys())[0]
            qrels_dict[qid][first_doc] = 1

    # 过滤掉 run 为空的 query
    valid_qids = [qid for qid in run_dict if run_dict[qid]]
    qrels_dict = {qid: qrels_dict[qid] for qid in valid_qids}
    run_dict = {qid: run_dict[qid] for qid in valid_qids}

    if not qrels_dict or not run_dict:
        return {"recall@5": 0, "ndcg@5": 0, "mrr": 0}

    qrels = Qrels(qrels_dict)
    run = Run(run_dict)
    results = evaluate(qrels, run, ["recall@5", "ndcg@5", "mrr"])
    return {k: float(v) for k, v in results.items()}

param_grid = {

    "chunk_overlap": [20, 50, 100],
    "top_k": [3, 5, 10],

best_params = None

    for co in param_grid["chunk_overlap"]:

            continue
        for tk in param_grid["top_k"]:
            metrics = evaluate_config(cs, co, tk)
            score = metrics["ndcg@5"]  # 用NDCG@5作为优化目标
            all_results.append({
                "chunk_size": cs, "chunk_overlap": co, "top_k": tk,
                **metrics,
            })
            print(f"  chunk_size={cs:<5} overlap={co:<4} top_k={tk:<3} | "
                  f"Recall@5={metrics['recall@5']:.4f} NDCG@5={metrics['ndcg@5']:.4f} MRR={metrics['mrr']:.4f}")
            if score > best_score:
                best_score = score
                best_params = {"chunk_size": cs, "chunk_overlap": co, "top_k": tk}

print(f"\n最优配置")

2. 每个事实必须标注来源编号 [1][2]，并优先引用相关度高的来源
3. 如果论文之间有矛盾，指出矛盾并分别陈述
4. 如果参考论文不足以回答，明确说明"现有论文不足以回答该问题"
5. 回答结构：直接结论 → 详细分析 → 引用论文列表

{context}

回答：""")

    if question.lower() == "q":
        break

    if not question:
        continue

    raw = multi_query_chain.invoke({"question": question, "count": 3})
    try:
        queries = json.loads(raw)
        if len(queries) == 0 or not isinstance(queries, list):
            queries = [question]
    except Exception as e:
        queries = [question]

    if question not in queries:
        queries.insert(0, question)

    # 对每个查询做混合检索,合并信息
    all_candidates = {}
    for q in queries:
        # 向量检索
        vec_results = vectorstore.similarity_search_with_score(q, k=25)
        vec_ranking = []
        # 遍历检索的结果,得到doc对象,找到该doc在全局chunks切片列表里的下标索引
        for doc, _ in vec_results:
            for i, c in enumerate(chunks):
                if c.page_content == doc.page_content:
                    vec_ranking.append(i)
                    break
        # BM检索
        bm25_scores = bm25.get_scores(list(jieba.cut(q)))
        bm25_rank = list(np.argsort(bm25_scores)[::-1][:25])

        # RRF融合
        fused = {}
        for ranking in [vec_ranking, bm25_rank]:
            for rank, idx in enumerate(ranking):
                fused[idx] = fused.get(idx, 0) + 1 / (60 + rank + 1)

        # 多查询合并候选集,去重
        for idx, score in fused.items():
            # key 文本 value元组 (文档对象,累计总分数)
            key = chunks[idx].page_content
            if key not in all_candidates:
                all_candidates[key] = (chunks[idx], 0)

            all_candidates[key] = (chunks[idx], all_candidates[key][1] + score)
    candidates = sorted(all_candidates.values(), key=lambda x: x[1], reverse=True)
    print(f"[混合检索] 多查询合并后,共 {len(candidates)} 个唯一候选")

    # 重排序
    pairs = [[question, chunk.page_content] for chunk, _ in candidates]
    reranker = CrossEncoder("./bge-reranker-v2-m3")
    rerank_scores = reranker.predict(pairs)
    # 粗排+精排的格式: 文档元组 重排分数  ((chunk,rrf_score),rerank_scores) ((Doc,0.85),0.93)
    ranked = sorted(zip(candidates, rerank_scores), key=lambda x: x[1], reverse=True)[:5]
    # 精排后的格式: (chunk, rerank_scores)  (Doc,0.93)
    reranked = [(chunk, float(s)) for (chunk, _), s in ranked]
    print(f"[重排序] {len(candidates)} 个候选")

    # 构建上下文+生成回答
    context_parts = []
    for i, (doc, score) in enumerate(reranked, 1):
        context_parts.append(
            f"[{i}] [相关度: {score}]"
            f"标题:{doc.metadata['title']}"
            f"日期:{doc.metadata['published']}"
            f"内容:{doc.page_content}"
        )

    context = "\n\n".join(context_parts)
    answer = answer_chain.invoke({"context": context, "question": question})
    print(f"\n {answer} \n")
    for i, (doc, score) in enumerate(reranked, 1):
        print(
            f"[{i}] [相关度: {score}]"
            f"标题:{doc.metadata['title']}"
            f"日期:{doc.metadata['published']}"
            f"内容:{doc.page_content}"
        )

from rank_bm25 import BM25Okapi
import jieba
from sentence_transformers import SentenceTransformer
from sklearn.metrics.pairwise import cosine_similarity
import numpy as np

# 文档
documents = [
    "RAG通过检索外部知识库增强大语言模型的回答能力",
    "向量数据库用于高效存储和检索高维向量，支持近似最近邻搜索",

    "Python是一种高级编程语言，广泛用于数据科学和机器学习",
]
query = "RAG如何结合BM25进行混合检索"

# BM25检索

tokenized_query = list(jieba.cut(query))
bm_scores = bm25.get_scores(tokenized_query)

# 向量检索
model = SentenceTransformer("./bge-base-zh-v1.5")
query_vec = model.encode([query])
doc_vec = model.encode(documents)
vec_sims = cosine_similarity(query_vec, doc_vec)[0]
vec_rank = np.argsort(vec_sims)[::-1]

# RRF融合
rrf_scores = np.zeros(len(documents))
for rank, doc_idx in enumerate(bm25_rank):
    rrf_scores[doc_idx] += 1 / (60 + rank + 1)

for rank, doc_idx in enumerate(vec_rank):
    rrf_scores[doc_idx] += 1 / (60 + rank + 1)

fused_rank = np.argsort(rrf_scores)[::-1]
print(fused_rank)

# 打印
print(f"查询: {query}\n")
print(f"{'排名':<6} {'BM25':<20} {'向量':<20} {'RRF':<20}")
print("-" * 70)
for i in range(len(documents)):
    print(f"{i + 1:<6} "
          f"{documents[bm25_rank[i]][:10]:<20}"
          f"{documents[vec_rank[i]][:10]:<20}"
          f"{documents[fused_rank[i]][:10]:<20}")

    print(f"  {documents[idx][:10]}:{rrf_scores[idx]}")

    return (scores - scores.min()) / (scores.max() - scores.min() + 1e-10)

bm25_norm = ln(bm_scores)
vec_norm = ln(vec_sims)

wlc_scores = bm25_norm * .5 + vec_norm * .5

"""
不同算法的索引对比
"""
import numpy as np
import faiss
import time

np.random.seed(42)
N = 100000
d = 768
K = 10

database_vectors = np.random.randn(N, d).astype("float32")
faiss.normalize_L2(database_vectors)
query_vectors = np.random.randn(100, d).astype("float32")  # 100个查询取平均
faiss.normalize_L2(query_vectors)

# Ground Truth
index_flat_gt = faiss.IndexFlatL2(d)
index_flat_gt.add(database_vectors)
_, gt_indices = index_flat_gt.search(query_vectors, K)

def evaluate_index(index, name, train_data=None):
    # 训练（如果需要）
    if train_data is not None and not index.is_trained:
        start = time.time()
        index.train(train_data)
        train_t = time.time() - start
    else:
        train_t = 0

    # 构建
    start = time.time()
    index.add(database_vectors)
    build_t = time.time() - start

    # 查询
    start = time.time()
    _, pred_indices = index.search(query_vectors, K)
    query_t = (time.time() - start) * 1000 / len(query_vectors)

    # 召回率
    recalls = []
    for i in range(len(query_vectors)):
        recall = len(set(gt_indices[i]) & set(pred_indices[i])) / K
        recalls.append(recall)
    avg_recall = np.mean(recalls)

    return {
        "name": name,
        "train_time": train_t,
        "build_time": build_t,
        "query_time_ms": query_t,
        "recall": avg_recall,
    }

# 测试四种索引
results = []

# 1. Flat
results.append(evaluate_index(faiss.IndexFlatL2(d), "Flat"))

# 2. HNSW
idx = faiss.IndexHNSWFlat(d, 32)
idx.hnsw.efConstruction = 200
idx.hnsw.efSearch = 100
results.append(evaluate_index(idx, "HNSW(M=32,ef=100)"))

# 3. IVF
quantizer = faiss.IndexFlatL2(d)
idx = faiss.IndexIVFFlat(quantizer, d, 256)
idx.nprobe = 16
results.append(evaluate_index(idx, "IVF(nlist=256,nprobe=16)", database_vectors))

# 4. LSH
idx = faiss.IndexLSH(d, 128)
results.append(evaluate_index(idx, "LSH(128bit)", database_vectors))

# 打印对比
print(f"{'算法':<25} {'训练(s)':<10} {'构建(s)':<10} {'查询(ms)':<12} {'召回率@10':<12}")
print("-" * 70)
for r in results:
    print(f"{r['name']:<25} {r['train_time']:<10.2f} {r['build_time']:<10.2f} "
          f"{r['query_time_ms']:<12.2f} {r['recall']:<12.2%}")

```
"""
文本的综合清洗
"""
import os
import re
import hashlib
import unicodedata
from bs4 import BeautifulSoup
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity
import numpy as np
from collections import defaultdict
from langchain_community.document_loaders import TextLoader, PyMuPDFLoader, Docx2txtLoader
from config import *

os.makedirs(OUTPUT_DIR, exist_ok=True)

# 文档加载
LOADERS = {
    ".pdf": lambda f: "\n".join(d.page_content for d in PyMuPDFLoader(f).load()),
    ".docx": lambda f: "\n".join(d.page_content for d in Docx2txtLoader(f).load()),
    ".txt": lambda f: TextLoader(f, encoding="utf-8").load()[0].page_content,
    ".md": lambda f: TextLoader(f, encoding="utf-8").load()[0].page_content
}
raw_texts = {}
for filename in sorted(os.listdir(INPUT_DIR)):
    filepath = os.path.join(INPUT_DIR, filename)
    if os.path.isfile(filepath):
        ext = os.path.splitext(filename)[1].lower()
        if ext in LOADERS:
            raw_texts[filename] = LOADERS[ext](filepath)

# 基础清洗
def clean_text(text):
    # HTML清洗
    while re.search(HTML_TAG_PATTERN, text):
        text = BeautifulSoup(text, "html.parser").get_text(separator="\n")

    # unicode归一化
    text = unicodedata.normalize("NFKC", text)

    # 统一换行
    text = text.replace("\r\n", "\n").replace("\r", "\n")

    # 过滤 C类控制字符
    text = "".join(c for c in text if unicodedata.category(c)[0] != "C" or c in "\n\t")

    # 乱码
    for pattern, _ in GARBLED_PATTERNS:
        text = re.sub(pattern, "", text)

    # 压缩空格
    text = "\n".join(re.sub(r' {2,}', ' ', line).strip() for line in text.split("\n"))

    # 压缩连续换行
    text = re.sub(r"\n{3,}", "\n\n", text)

    # 页眉+页脚+水印
    for pattern in HEADER_FOOTER_PATTERNS + WATERMARK_PATTERNS:
        text = re.sub(pattern, '', text, flags=re.IGNORECASE | re.MULTILINE)

    # Markdown符号
    for pattern, replacement, user_multiline, _ in MARKDOWN_PATTERNS:
        text = re.sub(pattern, replacement, text, flags=re.MULTILINE if user_multiline else 0)

    text = text.replace(*MARKDOWN_INLINE_PIPE)

    # 孤立的页码,无效的分割线
    for pattern in NOISE_LINE_PATTERNS:
        text = re.sub(pattern, "", text, flags=re.MULTILINE)

    return re.sub(r"\n{3,}", "\n\n", text).strip()

cleaed_texts = {fn: clean_text(t) for fn, t in raw_texts.items()}

# 段落切分
all_paragraphs = [
    (fn, para.strip())
    for fn, t in cleaed_texts.items()
    for para in t.split("\n")
    if len(para.strip()) >= MIN_PARAGRAPH_LENGTH
]

# MD5去重
seen = set()
after_hash = []
for source, para in all_paragraphs:
    h = hashlib.md5(re.sub(r'\s+', '', para).encode("utf-8")).hexdigest()

    if h not in seen:
        seen.add(h)
        after_hash.append((source, para))
hash_removed = len(all_paragraphs) - len(after_hash)

# 模糊去重
final_paragraphs = after_hash
fuzzy_removed = 0
if len(after_hash) > 1:
    sources = [s for s, _ in after_hash]  # [1.pdf ,2.txt...]
    paras = [p for _, p in after_hash]  # [文本1,文本2...]
    vec = TfidfVectorizer(analyzer="char", ngram_range=(2, 3), max_features=20000)
    sim = cosine_similarity(vec.fit_transform(paras))

    keep = np.ones(len(paras), dtype=bool)
    for i in range(len(paras)):
        if keep[i]:
            keep[i + 1:] &= sim[i, i + 1:] < SIMILARITY_THRESHOLD

    final_paragraphs = [(sources[i], paras[i]) for i in range(len(paras)) if keep[i]]

    fuzzy_removed = len(after_hash) - len(final_paragraphs)

# 质量的校验
total_chars = sum(len(p) for _, p in final_paragraphs)
short_count = sum(1 for _, p in final_paragraphs if len(p) < MIN_PARAGRAPH_LENGTH)
long_count = sum(1 for _, p in final_paragraphs if len(p) > MAX_PARAGRAPH_LENGTH)
normal_ratio = (len(final_paragraphs) - short_count - long_count) / len(final_paragraphs) * 100
noise_count = sum(
    1 for _, p in final_paragraphs
    if any(re.search(pat, p) for pat in NOISE_LINE_PATTERNS)
)
noise_rate = noise_count / len(final_paragraphs) * 100

print(
    f"总字数量:{total_chars},短段落:{short_count},长段落{long_count},正常段落占比:{normal_ratio},噪声段落占比:{noise_rate}")

# 保存
groped = defaultdict(list)
for source, para in final_paragraphs:
    groped[source].append(para)

for source, paras in groped.items():
    basename = os.path.splitext(source)[0]
    with open(os.path.join(OUTPUT_DIR, f"{basename}_cleaned.txt"), "w", encoding="utf-8") as f:
        f.write("\n\n".join(paras) + "\n\n")

"""

import os
import re

INPUT_DIR = "./data_one"
OUTPUT_DIR = "./cleaned_one"
os.makedirs(OUTPUT_DIR, exist_ok=True)

# 读取
def ext_text(filepath):
    ext = os.path.splitext(filepath)[1].lower()
    if ext == ".pdf":
        import pdfplumber
        with pdfplumber.open(filepath) as pdf:
            return "\n".join(page.extract_text() or "" for page in pdf.pages)

    if ext in [".doc", ".docx"]:
        from docx import Document
        return "\n".join(p.text for p in Document(filepath).paragraphs)

    if ext in [".txt", ".md", ".csv"]:
        with open(filepath, "r", encoding="utf-8", errors="ignore") as f:
            return f.read()

# 清洗
def clean_text(raw):
    if not raw:
        return ""

    # 统一换行 制表符 控制字符
    text = raw.replace("\r\n", "\n").replace("\r", "\n").replace("\t", " ")
    text = re.sub(r'[\x00-\x08\x0b\x0c\x0e-\x1f\x7f]', "", text)

    # 合并多个空格 + NBSP + strip
    text = "\n".join(
        re.sub(r' {2,}', ' ', re.sub(r'\u00a0+', ' ', line)).strip()
        for line in text.split("\n")
    )

    # 乱码 页眉/脚 水印  休息+练习 10:25继续
    patterns = '|'.join([
        r'锟斤拷',
        r'烫+',
        r'屯+',
        r'鑴曪絿缇庯紝',
        r'\ufffd+',
        r'[\ufffe\uffff]',
        r'【内部机密】.*?编制',
        r'第\s*\d+\s*页\s*(共\s*\d+\s*页)?',
        r'Page\s+\d+\s*(of\s+\d+)?',
        r'[-—]\s*\d+\s*[-—]',
        r'版权所有\s*翻印必究',
        r'.*版本号：\s*V[\d.]+.*编制',
        r'^内部机密$', r'^.*CONFIDENTIAL.*$',
        r'^.*DRAFT.*$', r'^.*仅供内部使用.*$',
    ])
    text = re.sub(patterns, "", text, flags=re.MULTILINE | re.IGNORECASE)

    # 空行 + 去strip
    return re.sub(r'\n{3,}', '\n\n', text).strip()

files = sorted(f for f in os.listdir(INPUT_DIR)
               if os.path.isfile(os.path.join(INPUT_DIR, f))
               )

def expand_story(context):
    prompt = f"""你是一个创意故事作家。
    当前故事上下文：
    {context}

    请基于以上上下文，构思3个完全不同的情节发展方向，每个方向写2-3句话。

    3. 不要写完整结局，只写下一步的发展

    {{
      "branches": [

        {{"direction": "方向2的简短描述", "content": "续写的2-3句话内容"}},

    }}
    """

        model="deepseek-chat",
        messages=[{"role": "user", "content": prompt}],
        temperature=0.9,
        max_tokens=1024,
    )

    return parse_json(output)["branches"]

    2. 连贯性：是否自然承接上文，人物行为是否合理
    3. 创意性：是否新颖不俗套

    严格按以下JSON格式返回：
    {{
      "attraction": 吸引力分数,
      "coherence": 连贯性分数,
      "creativity": 创意性分数,

        model="deepseek-chat",

def parse_json(text):
    try:
        return json.loads(text)
    except json.JSONDecodeError:
        if "```json" in text:
            text = text.split("```json")[1].split("```")[0].strip()
        elif "```" in text:
            text = text.split("```")[1].split("```")[0].strip()

        return json.loads(text)

			# 思维树主算法  逐层扩展 - 评估 - 剪枝(束搜索) xa345e6r7ty897u4y3521432q1`2W3ETFH VN
def TOT(start_text, max_depth=3, beam_width=2):
    print("思维树创意故事生成~~~~")
    print(f"故事开头:{start_text}")

    # 初始化前沿节点
    # {"context": 故事文本, "path": 经过的分支列表, "score": 累计分}
    frontier = [{"context": start_text, "path": [], "score": 0}]

    for depth in range(max_depth):
        print(f"\n第{depth + 1}层: 当前{len(frontier)}个分支待扩展")

        # 保留下一层所有候选节点
        all_candidates = []
        for node_idx, node in enumerate(frontier):
            context = node["context"]
            print(f"📄 分支: {node_idx + 1},累计:{node['score']:.1f}")

            print("🔎 LLM扩展...")
            branches = expand_story(context)

            # 对每个分支评估
            for branch in branches:
                print(f"候选的方向:{branch['direction']}")
                print(f"  内容:{branch['content'][:50]}...")

                eval_result = evaluate(context, branch)
                score = eval_result.get("score", 5)
                reason = eval_result.get("reason", "")

                print(f"  评分:{score}/10"
                      f"吸引力:{eval_result.get('attraction', 0)} "
                      f"连贯性:{eval_result.get('coherence', 0)} "
                      f"创意性:{eval_result.get('creativity', 0)} ")
                print(f"  理由:{reason}")

                new_node = {
                    "context": context + "\n\n" + branch["content"],
                    "path": node["path"] + [branch],
                    "score": node["score"] + score
                }

                all_candidates.append(new_node)

        # 剪枝
        all_candidates.sort(key=lambda x: x["score"], reverse=True)
        frontier = all_candidates[:beam_width]
        print(f" ✂️ 剪枝后保留{len(frontier)}个最优分支")

        for i, node in enumerate(frontier, start=1):
            print(f"    分支:{i}:累计分:{node['score']:.1f}")

    # 保留分数最高的
    best = max(frontier, key=lambda x: x["score"])

    return best["context"]

from langchain_openai import ChatOpenAI
from langchain.tools import tool

from langchain.agents import AgentExecutor, create_tool_calling_agent

load_dotenv()

# 1.定义工具函数
# @tool把普通的函数变成Langchain工具
@tool
def get_weather(city: str) -> dict:
    """
    查询指定城市的天气信息
    参数:
        city: 城市名称
    返回:
        包含温度和天气状况的字典
    """
    weather_db = {
        "北京": {"temp": 26, "condition": "下雨", "humidity": 80},
        "上海": {"temp": 32, "condition": "晴", "humidity": 60},
        "广州": {"temp": 30, "condition": "多云", "humidity": 73}
    }
    return weather_db.get(city, {"temp": "未知"})

@tool
def get_time() -> str:
    """
    获取当前时间,不需要参数
    """
    from datetime import datetime
    return datetime.now().strftime("%d.%m.%Y %H:%M:%S")

tools = [get_weather, get_time]

# 2.初始化大模型
llm = ChatOpenAI(
    model="deepseek-chat",
    api_key=os.getenv("DEEPSEEK_API_KEY"),
    base_url=os.getenv("DEEPSEEK_BASE_URL"),
)

    ("system", "你是一个智能助手,可以使用工具查询天气,回答简洁明了"),
    ("user", "{input}"),
    MessagesPlaceholder(variable_name="agent_scratchpad"),
])

# 4.agent创建

    agent=agent, tools=tools, verbose=True, max_iterations=5

    return result["output"]

```
"""
推测解码
小模型"猜"，大模型"验证"，在不损失质量的前提下加速推理
"""

import torch
import time 
from transformers import GPT2Tokenizer, GPT2LMHeadModel

# 第一步：加载大模型和小模型
device = "cuda" if torch.cuda.is_available() else "cpu"

tokenizer = GPT2Tokenizer.from_pretrained("./gpt2")
tokenizer.pad_token = tokenizer.eos_token

# 小模型（draft model）：distilgpt2，82M参数，6层，推理快
draft_model = GPT2LMHeadModel.from_pretrained("./distilgpt2")
draft_model = draft_model.to(device)
draft_model.eval()

# 大模型（target model）：gpt2，124M参数，12层，质量高
target_model = GPT2LMHeadModel.from_pretrained("./gpt2")
target_model = target_model.to(device)
target_model.eval()

# 第二步：标准自回归生成（基准，用于对比）
def standard_generation(model, input_ids, max_new_tokens, temperature=1.0):
    generated = input_ids.clone()
    with torch.no_grad():
        for _ in range(max_new_tokens):
            outputs = model(input_ids=generated)
            logits = outputs.logits[:, -1, :] / temperature
            probs = torch.softmax(logits, dim=-1)
            next_token = torch.multinomial(probs, num_samples=1)
            generated = torch.cat([generated, next_token], dim=-1)
    return generated

# 第三步：推测解码实现
def speculative_decoding(draft_model, target_model, input_ids, max_new_tokens, gamma=4, temperature=1.0):
    generated = input_ids.clone()
    total_accepted = 0
    total_bonus = 0
    total_rounds = 0

    with torch.no_grad():
        while generated.shape[1] - input_ids.shape[1] < max_new_tokens:
            total_rounds += 1
            orig_len = generated.shape[1]

            # Step 1: 小模型猜gamma个
            draft_tokens = []
            draft_probs_list = []
            draft_seq = generated.clone()
            for _ in range(gamma):
                draft_outputs = draft_model(input_ids=draft_seq)
                draft_logits = draft_outputs.logits[:, -1, :] / temperature
                draft_probs = torch.softmax(draft_logits, dim=-1)
                draft_token = torch.multinomial(draft_probs, num_samples=1)
                draft_tokens.append(draft_token)
                draft_probs_list.append(draft_probs)
                draft_seq = torch.cat([draft_seq, draft_token], dim=-1)

            # Step 2: 大模型验证
            verify_input = torch.cat([generated] + draft_tokens, dim=-1)
            target_outputs = target_model(input_ids=verify_input)
            target_logits = target_outputs.logits / temperature

            # Step 3: 逐个验证
            n_accepted = 0
            for i in range(gamma):
                draft_probs_i = draft_probs_list[i]
                q = draft_probs_i[0, draft_tokens[i].item()].item()

                target_logits_i = target_logits[:, orig_len - 1 + i, :]
                target_probs_i = torch.softmax(target_logits_i, dim=-1)
                p = target_probs_i[0, draft_tokens[i].item()].item()

                accept_prob = min(1.0, p / q) if q > 0 else 1.0
                if torch.rand(1).item() < accept_prob:
                    generated = torch.cat([generated, draft_tokens[i]], dim=-1)
                    n_accepted += 1
                else:
                    adjusted_probs = torch.clamp(target_probs_i - draft_probs_i, min=0)
                    if adjusted_probs.sum() > 0:
                        adjusted_probs = adjusted_probs / adjusted_probs.sum()
                        corrected_token = torch.multinomial(adjusted_probs, num_samples=1)
                    else:
                        corrected_token = torch.multinomial(target_probs_i, num_samples=1)
                    generated = torch.cat([generated, corrected_token], dim=-1)
                    break

            # Step 4: 全接受时额外bonus token
            if n_accepted == gamma:
                bonus_logits = target_logits[:, -1, :]
                bonus_probs = torch.softmax(bonus_logits, dim=-1)
                bonus_token = torch.multinomial(bonus_probs, num_samples=1)
                generated = torch.cat([generated, bonus_token], dim=-1)
                total_bonus += 1

            total_accepted += n_accepted

            if generated.shape[1] - input_ids.shape[1] >= max_new_tokens:
                break

    # 接受率 = 被接受的猜测token / 总猜测token
    accept_rate = total_accepted / (total_rounds * gamma) if total_rounds > 0 else 0
    return generated, accept_rate, total_rounds, total_bonus

# 第四步：运行对比
print("对比测试：标准生成 vs 推测解码")

prompt = "The future of artificial intelligence is"
input_ids = tokenizer.encode(prompt, return_tensors="pt").to(device)
max_new_tokens = 80
gamma = 4

# 标准生成
print("[1/2] 标准自回归生成...")
start = time.time()
output_standard = standard_generation(target_model, input_ids, max_new_tokens, temperature=0.8)
time_standard = time.time() - start
text_standard = tokenizer.decode(output_standard[0], skip_special_tokens=True)
print(f"  耗时: {time_standard:.3f}s")
print(f"  生成: {text_standard[:100]}...")

# 推测解码
print("[2/2] 推测解码...")
start = time.time()
output_spec, accept_rate, rounds,bonus = speculative_decoding(
    draft_model, target_model, input_ids, max_new_tokens, gamma=gamma, temperature=0.8
)
time_spec = time.time() - start
text_spec = tokenizer.decode(output_spec[0], skip_special_tokens=True)
print(f"  耗时: {time_spec:.3f}s")
print(f"  接受率: {accept_rate:.1%}")
print(f"  总轮数: {rounds}（每轮猜{gamma}个，平均每轮确认 {accept_rate * gamma:.2f} 个token）")
print(f"  生成: {text_spec[:100]}...")

# 速度对比
speedup = time_standard / time_spec if time_spec > 0 else 0
print(f"  标准生成: {time_standard:.3f}s")
print(f"  推测解码: {time_spec:.3f}s")
print(f"  加速比: {speedup:.2f}x")
print(f"  接受率: {accept_rate:.1%}")

# 第五步：不同 gamma 值的影响
print("\n不同 gamma 值的影响")

for gamma in [2, 4, 6, 8]:
    start = time.time()
    _, accept_rate, rounds,bonus = speculative_decoding(
        draft_model, target_model, input_ids, max_new_tokens, gamma=gamma, temperature=0.8
    )
    elapsed = time.time() - start
    print(f"  gamma={gamma:2d}: 接受率={accept_rate:.1%}, 轮数={rounds:3d}, 耗时={elapsed:.3f}s")
```

# 手动实现GPT

```
"""
从0实现GPT
#   - 因果掩码（Causal Mask）
#   - 多头自注意力（Multi-Head Self-Attention）
#   - 前馈网络（Feed-Forward Network）
#   - 残差连接 + LayerNorm（Pre-LN）
#   - 词嵌入 + 可学习位置编码
#   - 语言建模头 + 自回归生成
"""

import torch
import torch.nn as nn
import torch.nn.functional as F
import urllib.request
import os

# 超参数
batch_size = 32  # 每批32个序列
block_size = 128  # 上下文窗口，数据量大了可以调大
n_layer = 6  # 6层Decoder
n_head = 8  # 12个注意力头
n_embd = 256  # 隐藏维度256
dropout = 0.1  # dropout比例，训练时设0.1防过拟合
learning_rate = 3e-4  # 学习率
max_iters = 10000  # 训练迭代次数，数据量大了可以多训
eval_interval = 500  # 每500步评估一次
eval_iters = 100  # 评估时跑100批取平均
device = "cuda" if torch.cuda.is_available() else "cpu"

# 1.准备数据（完整莎士比亚全集，约100万字符，字符级语言模型）
data_path = "shakespeare_full.txt"
# 如果本地没有就自动下载
if not os.path.exists(data_path):
    print("正在下载莎士比亚全集...")
    url = "https://raw.githubusercontent.com/karpathy/char-rnn/master/data/tinyshakespeare/input.txt"
    urllib.request.urlretrieve(url, data_path)
    print("下载完成！")

with open(data_path, "r", encoding="utf-8") as f:
    text = f.read()  # 读取完整文本

chars = sorted(list(set(text)))  # 所有出现过的字符，排序
vocab_size = len(chars)  # 词表大小（字符数）
print(f"词表大小（字符数）: {vocab_size}")
print(f"文本总长度: {len(text)} 字符")

# 字符和编号的映射
stoi = {ch: i for i, ch in enumerate(chars)}  # 字符 -> 编号
itos = {i: ch for i, ch in enumerate(chars)}  # 编号 -> 字符
encode = lambda s: [stoi[c] for c in s]  # 编码函数：字符串转编号列表
decode = lambda l: "".join([itos[i] for i in l])  # 解码函数：编号列表转字符串

# 全部文本转成张量
data = torch.tensor(encode(text), dtype=torch.long)  # 形状 [总长度]
n = int(0.9 * len(data))  # 90%训练集
train_data = data[:n]  # 训练集
val_data = data[n:]  # 验证集

print(f"训练集长度: {len(train_data)}, 验证集长度: {len(val_data)}")
print(f"block_size: {block_size}, 验证集够不够: {len(val_data) > block_size}")

# 取一批数据的函数
def get_batch(split):
    data = train_data if split == "train" else val_data  # 选训练或验证
    ix = torch.randint(len(data) - block_size, (batch_size,))  # 随机选batch_size个起始位置
    x = torch.stack([data[i:i + block_size] for i in ix])  # 输入：每个起始位置开始的block_size个字符
    y = torch.stack([data[i + 1:i + block_size + 1] for i in ix])  # 目标：输入右移一位（预测下一个字符）
    x, y = x.to(device), y.to(device)  # 移到设备
    return x, y

# 2.定义模型组件
# 组件1：因果自注意力头
class Head(nn.Module):
    def __init__(self, head_size):
        super().__init__()
        self.head_size = head_size  # 保存head_size，forward里缩放用
        self.key = nn.Linear(n_embd, head_size, bias=False)  # Key投影矩阵，无偏置
        self.query = nn.Linear(n_embd, head_size, bias=False)  # Query投影矩阵
        self.value = nn.Linear(n_embd, head_size, bias=False)  # Value投影矩阵
        # 注册因果掩码为buffer（不是参数，但随模型保存和移动设备）
        self.register_buffer("tril", torch.tril(torch.ones(block_size, block_size)))
        self.dropout = nn.Dropout(dropout)  # dropout层

    def forward(self, x):
        B, T, C = x.shape  # B=batch, T=序列长度, C=通道维度
        k = self.key(x)  # [B, T, head_size]，每个位置的Key
        q = self.query(x)  # [B, T, head_size]，每个位置的Query
        # 计算注意力分数：Q·K^T / sqrt(head_size)
        wei = q @ k.transpose(-2, -1) * (self.head_size ** -0.5)  # [B, T, T]，缩放点积
        # 应用因果掩码：把上三角设为-inf，这样Softmax后为0
        wei = wei.masked_fill(self.tril[:T, :T] == 0, float("-inf"))
        wei = F.softmax(wei, dim=-1)  # Softmax归一化，每行和为1
        wei = self.dropout(wei)  # dropout随机丢弃
        v = self.value(x)  # [B, T, head_size]，每个位置的Value
        out = wei @ v  # [B, T, head_size]，加权求和Value
        return out

# 组件2：多头注意力（把多个头拼起来）
class MultiHeadAttention(nn.Module):
    def __init__(self, num_heads, head_size):
        super().__init__()
        # 创建num_heads个独立的注意力头
        self.heads = nn.ModuleList([Head(head_size) for _ in range(num_heads)])
        self.proj = nn.Linear(n_embd, n_embd)  # 输出投影矩阵W^O
        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        # 每个头独立计算，然后在特征维度拼接
        out = torch.cat([h(x) for h in self.heads], dim=-1)  # [B, T, num_heads*head_size] = [B, T, n_embd]
        out = self.proj(out)  # 输出投影
        out = self.dropout(out)
        return out

# 组件3：前馈网络（FFN）
class FeedForward(nn.Module):
    def __init__(self, n_embd):
        super().__init__()
        # GPT用4倍升维，GELU激活
        self.net = nn.Sequential(
            nn.Linear(n_embd, 4 * n_embd),  # 升维4倍：128 -> 512
            nn.GELU(),  # GELU激活（GPT用GELU不用ReLU）
            nn.Linear(4 * n_embd, n_embd),  # 降维回来：512 -> 128
            nn.Dropout(dropout),
        )

    def forward(self, x):
        return self.net(x)

# 组件4：Transformer Block（一层Decoder）
class Block(nn.Module):
    def __init__(self, n_embd, n_head):
        super().__init__()
        head_size = n_embd // n_head  # 每个头的维度 = 总维度 / 头数
        self.sa = MultiHeadAttention(n_head, head_size)  # 多头自注意力
        self.ffwd = FeedForward(n_embd)  # 前馈网络
        self.ln1 = nn.LayerNorm(n_embd)  # 第一个LayerNorm（Pre-LN）
        self.ln2 = nn.LayerNorm(n_embd)  # 第二个LayerNorm（Pre-LN）

    def forward(self, x):
        # Pre-LN结构：先LayerNorm，再注意力，再加残差
        x = x + self.sa(self.ln1(x))  # 残差连接：x + 注意力(LayerNorm(x))
        x = x + self.ffwd(self.ln2(x))  # 残差连接：x + FFN(LayerNorm(x))
        return x

# 3.定义完整 GPT 模型
class GPT(nn.Module):
    def __init__(self):
        super().__init__()
        # 词嵌入表：每个字符对应一个n_embd维向量
        self.token_embedding_table = nn.Embedding(vocab_size, n_embd)
        # 位置编码：每个位置对应一个n_embd维向量（可学习，不是正弦）
        self.position_embedding_table = nn.Embedding(block_size, n_embd)
        # 堆叠n_layer个Transformer Block
        self.blocks = nn.Sequential(*[Block(n_embd, n_head) for _ in range(n_layer)])
        # 最后一层的LayerNorm（GPT-2在最后加了额外的LN）
        self.ln_f = nn.LayerNorm(n_embd)
        # 语言建模头：把n_embd维映射到vocab_size维，预测下一个字符
        self.lm_head = nn.Linear(n_embd, vocab_size)

    def forward(self, idx, targets=None):
        B, T = idx.shape  # B=batch, T=序列长度
        # token嵌入 + 位置嵌入
        tok_emb = self.token_embedding_table(idx)  # [B, T, n_embd]
        pos_emb = self.position_embedding_table(torch.arange(T, device=device))  # [T, n_embd]
        x = tok_emb + pos_emb  # 逐元素相加，[B, T, n_embd]
        # 过所有Transformer Block
        x = self.blocks(x)  # [B, T, n_embd]
        x = self.ln_f(x)  # 最后的LayerNorm
        logits = self.lm_head(x)  # [B, T, vocab_size]，每个位置预测下一个字符的分数

        if targets is None:
            loss = None  # 推理时不需要计算损失
        else:
            B, T, C = logits.shape
            logits = logits.view(B * T, C)  # 展平成 [B*T, vocab_size]
            targets = targets.view(B * T)  # 展平成 [B*T]
            loss = F.cross_entropy(logits, targets)  # 交叉熵损失

        return logits, loss

    # 自回归生成函数
    def generate(self, idx, max_new_tokens, temperature=0.7):
        for _ in range(max_new_tokens):  # 循环生成每个token
            idx_cond = idx[:, -block_size:]  # 只取最后block_size个token（截断上下文）
            logits, loss = self(idx_cond)  # 前向传播
            logits = logits[:, -1, :]  # 取最后一个位置的logits，[B, vocab_size]
            logits = logits / temperature
            probs = F.softmax(logits, dim=-1)  # Softmax转概率
            idx_next = torch.multinomial(probs, num_samples=1)  # 按概率采样1个token
            idx = torch.cat([idx, idx_next], dim=1)  # 拼到序列末尾
        return idx

# 4.训练模型
print("开始训练~~~")
model = GPT()  # 创建模型
model = model.to(device)  # 移到设备
print(f"模型参数量: {sum(p.numel() for p in model.parameters()) / 1e3:.1f}K")

optimizer = torch.optim.AdamW(model.parameters(), lr=learning_rate)

# 评估函数：在训练集和验证集上算平均损失
@torch.no_grad()  # 不计算梯度
def estimate_loss():
    out = {}
    model.eval()  # 评估模式
    for split in ["train", "val"]:
        losses = torch.zeros(eval_iters)
        for k in range(eval_iters):
            X, Y = get_batch(split)
            logits, loss = model(X, Y)
            losses[k] = loss.item()
        out[split] = losses.mean()
    model.train()  # 切回训练模式
    return out

# 训练循环
for iter in range(max_iters):
    # 定期评估
    if iter % eval_interval == 0:
        losses = estimate_loss()
        print(f"step {iter:4d}: train_loss={losses['train']:.4f}, val_loss={losses['val']:.4f}")

    xb, yb = get_batch("train")  # 取一批训练数据
    logits, loss = model(xb, yb)  # 前向传播，算损失
    optimizer.zero_grad(set_to_none=True)  # 清空梯度
    loss.backward()  # 反向传播，算梯度
    optimizer.step()  # 更新参数

# 第五步：生成文本
print("训练完成! 生成文本~~")

context = torch.zeros((1, 1), dtype=torch.long, device=device)  # 起始输入：一个0（换行符）
generated = model.generate(context, max_new_tokens=500)  # 生成500个字符
generated_text = decode(generated[0].tolist())  # 解码成字符串
print(generated_text)

print("手动构造因果掩码，理解 GPT 如何实现'看不到未来'")

seq_len = len(tokens)  # 序列长度6

# 验证：模拟一次注意力计算
Q = torch.randn(seq_len, 64)  # 随机 Query，64维
K = torch.randn(seq_len, 64)  # 随机 Key，64维
scores = Q @ K.T / (64 ** 0.5)  # 注意力分数 = QK^T / sqrt(d_k)
scores_masked = scores + mask  # 加因果掩码
weights = torch.softmax(scores_masked, dim=-1)  # Softmax 归一化

print("加掩码后的注意力权重（上三角应该全是0）：")
print(weights.numpy().round(3))
print(f"每行和（应该都是1）: {weights.sum(dim=-1).numpy().round(3)}")

# GPT2-文本生成

```
"""
文本生成 - 理解自回归生成过程
"""
import torch
from transformers import GPT2Tokenizer, GPT2LMHeadModel

# 1.加载模型和分词器
device = "cuda" if torch.cuda.is_available() else "cpu"
print(f"使用设备: {device}")

# 加载 GPT-2 分词器，词表 50257
tokenizer = GPT2Tokenizer.from_pretrained("./gpt2")
model = GPT2LMHeadModel.from_pretrained("./gpt2")
model = model.to(device)
model.eval()

print(f"模型参数量: {sum(p.numel() for p in model.parameters()) / 1e6:.1f}M")

# 2.准备输入
prompt = "Artificial intelligence is"  # 提示文本，让 GPT-2 接着往下写
# 文本转 token，return_tensors="pt" 表示返回 PyTorch 张量
inputs = tokenizer(prompt, return_tensors="pt")
# input_ids 形状 [1, seq_len]，移到设备
input_ids = inputs["input_ids"].to(device)
# attention_mask 标记哪些位置是真实 token 哪些是 padding
attention_mask = inputs["attention_mask"].to(device)

print(f"输入文本: {prompt}")
print(f"输入 token IDs: {input_ids[0].tolist()}")
print(f"输入 token 数: {input_ids.shape[1]}")

# 3. 自回归生成
print("自回归生成...")

generated = input_ids.clone()  # 复制输入，作为生成的起始序列
max_new_tokens = 200

with torch.no_grad():
    for step in range(max_new_tokens):
        outputs = model(input_ids=generated, attention_mask=torch.ones_like(generated))
        # 取最后一个位置的 logits，形状 [1, 50257]，这就是下一个 token 的预测分数
        next_token_logits = outputs.logits[:, -1, :]
        next_token_id = torch.argmax(next_token_logits, dim=-1, keepdim=True)
        # 把新 token 拼到序列末尾，形状 [1, seq_len+1]
        generated = torch.cat([generated, next_token_id], dim=-1)

        # 实时打印生成过程
        # 把新 token 的数字转回文本
        new_token_text = tokenizer.decode(next_token_id[0])
        print(f"Step {step + 1:2d}: 预测 token_id={next_token_id.item():5d} -> 文本='{new_token_text}'")

# 把整个序列的 token 转回文本
generated_text = tokenizer.decode(generated[0])
print(f"最终生成: {generated_text}")

# 4.对比不同解码策略
print("不同解码策略对比...")

# 策略1：Greedy Search（贪心，每次选概率最高的）
output_greedy = model.generate(
    input_ids,  # 输入 token
    max_new_tokens=200,  # 最多生成30个新token
    do_sample=False,  # 不采样，就是贪心
    attention_mask=attention_mask,  # 注意力掩码
    pad_token_id=tokenizer.eos_token_id,  # padding 用 eos 填充，避免警告
)
print(f"\n[Greedy 贪心]   {tokenizer.decode(output_greedy[0], skip_special_tokens=True)}")

# 策略2：Beam Search（束搜索，保留多个候选路径）
output_beam = model.generate(
    input_ids,
    max_new_tokens=200,
    num_beams=5,  # 保留5条候选路径
    do_sample=False,
    attention_mask=attention_mask,
    pad_token_id=tokenizer.eos_token_id,
)
print(f"[Beam 束搜索]   {tokenizer.decode(output_beam[0], skip_special_tokens=True)}")

# 策略3：Top-K 采样（只在概率最高的 K 个 token 中随机采样）
output_topk = model.generate(
    input_ids,
    max_new_tokens=200,
    do_sample=True,  # 开启采样
    top_k=50,  # 只考虑概率最高的50个token
    temperature=0.7,  # 温度，小于1更确定，大于1更随机
    attention_mask=attention_mask,
    pad_token_id=tokenizer.eos_token_id,
)
print(f"[Top-K 采样]    {tokenizer.decode(output_topk[0], skip_special_tokens=True)}")

# 策略4：Top-P (Nucleus) 采样（在累积概率达到 P 的最小 token 集合中采样）
output_topp = model.generate(
    input_ids,
    max_new_tokens=200,
    do_sample=True,
    top_p=0.9,  # 累积概率达到0.9的最小集合
    temperature=0.7,
    attention_mask=attention_mask,
    pad_token_id=tokenizer.eos_token_id,
)
print(f"[Top-P 采样]    {tokenizer.decode(output_topp[0], skip_special_tokens=True)}")

# 5.混合策略 topk topp 重复惩罚 禁止3-gram重复
output_mix = model.generate(
    input_ids,
    max_new_tokens=200,
    do_sample=True,
    top_p=0.9,
    top_k=50,
    repetition_penalty=2.5,
    no_repeat_ngram_size=3,
    temperature=0.7,
    attention_mask=attention_mask,
    pad_token_id=tokenizer.eos_token_id,
)
print(f"[混合策略 ]    {tokenizer.decode(output_topp[0], skip_special_tokens=True)}")

# 6.理解 logits 和概率分布
print("最后一个位置的概率分布 Top-10")

with torch.no_grad():
    outputs = model(input_ids=input_ids)
    last_logits = outputs.logits[0, -1, :]
    probs = torch.softmax(last_logits, dim=-1)
    top10_probs, top10_ids = torch.topk(probs, k=10)

    for i, (pid, p) in enumerate(zip(top10_ids, top10_probs)):
        token_text = tokenizer.decode([pid.item()])
        print(f"  Top-{i + 1:2d}: token='{token_text:15s}' 概率={p.item():.4f} ({p.item() * 100:.1f}%)")
```

from transformers import BertTokenizer, BertModel
import torch
from sklearn.metrics.pairwise import cosine_similarity

tokenizer = BertTokenizer.from_pretrained('bert-base-chinese')
model = BertModel.from_pretrained('bert-base-chinese')
model.eval()

# def get_emb(sentence):
#     inputs = tokenizer(sentence, return_tensors='pt', padding=True, truncation=True)
#     with torch.no_grad():
#         outputs = model(**inputs)

#     cls_emb = outputs.last_hidden_state[:, 0, :]

def get_emb(sentence):
    inputs = tokenizer(sentence, return_tensors='pt', padding=True, truncation=True)
    with torch.no_grad():
        outputs = model(**inputs)

    att_mask = inputs["attention_mask"].unsqueeze(-1).expand(outputs.last_hidden_state.size()).float()
    sum_emb = torch.sum(outputs.last_hidden_state * att_mask, dim=1)
    sum_mask = torch.clamp(att_mask.sum(dim=1), min=1e-5)
    cls_emb = sum_emb / sum_mask
    return cls_emb.numpy()

v1 = get_emb("我喜欢Python编程")

v3 = get_emb("今天天气不错")
